import paramiko
import concurrent.futures

SMART_CMD = (
    "grep -E '^[[:space:]]*[0-9]+[[:space:]]+Power_On_Hours' "
    "/mnt/pss/smartctl_full_dump.log | tail -n 1 | awk '{print $10}'"
)

def get_power_on_hours(hostname, username, password):
    client = paramiko.SSHClient()
    client.set_missing_host_key_policy(paramiko.AutoAddPolicy())
    try:
        client.connect(hostname, username=username, password=password,
                       timeout=10, banner_timeout=30)
        stdin, stdout, stderr = client.exec_command(SMART_CMD, timeout=30)
        out = stdout.read().decode('utf-8').strip()
        return int(out) if out.isdigit() else None
    except Exception as e:
        print(f'Unable to get Power_On_Hours from {hostname}. Exception: {e}')
        return None
    finally:
        client.close()
